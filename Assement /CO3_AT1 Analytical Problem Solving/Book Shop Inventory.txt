#include <iostream>
#include <string>
#include <iomanip>
using namespace std;

class books
{
private:
    string author;
    string title;
    string publisher;
    double price;
    int stock;

    // Static members for transaction statistics
    static int successfulTransactions;
    static int unsuccessfulTransactions;

    // Private function to update price
    void updatePrice(double newPrice)
    {
        price = newPrice;
    }

public:

    // Constructor
    books(string a, string t, double p, string pub, int s)
    {
        author = a;
        title = t;
        price = p;
        publisher = pub;
        stock = s;
    }

    // Display book details
    void display()
    {
        cout << "\nBook Details" << endl;
        cout << "Author    : " << author << endl;
        cout << "Title     : " << title << endl;
        cout << "Publisher : " << publisher << endl;
        cout << "Price     : Rs. " << fixed << setprecision(2) << price << endl;
        cout << "Stock     : " << stock << endl;
    }

    // Search book and process transaction
    void purchase(string t, string a, int copies)
    {
        if (title == t && author == a)
        {
            cout << "\nBook is available." << endl;
            display();

            if (copies <= stock)
            {
                double total = copies * price;

                cout << "\nTotal Cost: Rs. " << total << endl;

                // Automatically update stock
                stock = stock - copies;

                successfulTransactions++;

                cout << "Transaction successful." << endl;
                cout << "Remaining Stock: " << stock << endl;
            }
            else
            {
                cout << "\nRequired copies not in stock" << endl;
                unsuccessfulTransactions++;
            }
        }
        else
        {
            cout << "\nBook not available." << endl;
            unsuccessfulTransactions++;
        }
    }

    // Public function to update price
    void changePrice(double newPrice)
    {
        updatePrice(newPrice);
        cout << "Price updated successfully." << endl;
    }

    // Display transaction statistics
    static void statistics()
    {
        cout << "\n----- Transaction Statistics -----" << endl;
        cout << "Successful Transactions   : "
             << successfulTransactions << endl;
        cout << "Unsuccessful Transactions : "
             << unsuccessfulTransactions << endl;
    }

    // Destructor
    ~books()
    {
        // Destructor
    }
};

// Definition of static members
int books::successfulTransactions = 0;
int books::unsuccessfulTransactions = 0;


int main()
{
    // Creating book objects using constructor
    books b1("R.K. Narayan", "Malgudi Days",
             450, "Indian Thought Publications", 10);

    books b2("Chetan Bhagat", "Five Point Someone",
             350, "Rupa Publications", 8);

    books b3("J.K. Rowling", "Harry Potter",
             600, "Bloomsbury", 5);

    string title, author;
    int copies;

    cout << "===== BOOK SHOP INVENTORY SYSTEM =====" << endl;

    // First transaction
    cout << "\nEnter book title: ";
    getline(cin, title);

    cout << "Enter author name: ";
    getline(cin, author);

    cout << "Enter number of copies required: ";
    cin >> copies;

    b1.purchase(title, author, copies);

    cin.ignore();

    // Second transaction
    cout << "\nEnter book title: ";
    getline(cin, title);

    cout << "Enter author name: ";
    getline(cin, author);

    cout << "Enter number of copies required: ";
    cin >> copies;

    b2.purchase(title, author, copies);

    // Updating price using private function indirectly
    cout << "\nUpdating price of Malgudi Days..." << endl;
    b1.changePrice(500);

    // Display statistics
    books::statistics();

    return 0;
}
